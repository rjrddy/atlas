cmake_minimum_required(VERSION 3.20)
project(atlas VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SANITIZER "" CACHE STRING "address or thread")
option(PYTHON_BINDINGS "Build Python module" ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
if(SANITIZER)
 add_compile_options(-fsanitize=${SANITIZER} -fno-omit-frame-pointer -g)
 add_link_options(-fsanitize=${SANITIZER})
endif()
find_package(Threads REQUIRED)
add_library(atlas_core src/index.cpp)
set_target_properties(atlas_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(atlas_core PUBLIC src)
add_executable(core_tests tests/core.cpp)
target_link_libraries(core_tests PRIVATE atlas_core Threads::Threads)
enable_testing()
add_test(NAME core COMMAND core_tests)
if(PYTHON_BINDINGS)
 find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
 execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
 list(APPEND CMAKE_PREFIX_PATH ${PYBIND_DIR})
 find_package(pybind11 CONFIG REQUIRED)
 pybind11_add_module(atlas_native src/bindings.cpp)
 target_link_libraries(atlas_native PRIVATE atlas_core)
endif()
